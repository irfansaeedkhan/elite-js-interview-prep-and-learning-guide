There are no automated tests for the async path: results arrive after the test block runs, so check the console output instead. Correct output is the user object for id 1, then Error: HTTP 404 for id 99, in that order."
  hint="Await the response, check ok before touching the body, and throw on a bad status. Returning the response itself is not enough: the caller wants the parsed JSON."

  const users = { 1: { id: 1, name: 'Ada' } };
  const id = Number(url.split('/').pop());
  if (users[id]) {
    return Promise.resolve({ ok: true, status: 200, json: () => Promise.resolve(users[id]) });
  }
  return Promise.resolve({ ok: false, status: 404, json: () => Promise.resolve({ message: 'not found' }) });
}

// your code here: define loadUser(id) using mockFetch

loadUser(1)
  .then((user) => console.log('user: ' + JSON.stringify(user)))
  .catch((e) => console.log('Error: ' + e.message))
  .then(() => loadUser(99))
  .then((user) => console.log('user: ' + JSON.stringify(user)))
  .catch((e) => console.log('Error: ' + e.message));`}
