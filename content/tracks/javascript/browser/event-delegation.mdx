  title="Delegate clicks from a parent to its buttons"
  prompt="The sandbox has no real DOM, so this exercise runs against a small node tree with the same semantics as the real thing: nodes have a tag and children, click() dispatches an event that bubbles from the clicked node up to the root, and each handler receives an event with target (where the click started) and currentTarget (the node handling it now). Write delegate(parent, tag, handler) so it attaches ONE listener to the parent and calls handler with the matching node only when the click started on, or inside, an element of that tag. Clicks elsewhere must not call it."
  hint="Listen once on the parent. From event.target, walk up through parents until you find the tag or reach the parent itself. A click on a span inside a button still counts as a button click."
  starterCode={`function makeNode(tag, children) {
  const node = {
    tag,
    parent: null,
    listeners: [],
    addEventListener(type, fn) { if (type === 'click') node.listeners.push(fn); },
    click() {
      let current = node;
      const event = { target: node, currentTarget: null };
      while (current) {
        event.currentTarget = current;
        current.listeners.forEach((fn) => fn(event));
        current = current.parent;
      }
    },
  };
  (children || []).forEach((child) => { child.parent = node; });
  return node;
}

const spanInButton = makeNode('span', []);
const button = makeNode('button', [spanInButton]);
const div = makeNode('div', []);
const root = makeNode('div', [button, div]);
let handled = [];
// your code here: define delegate, then delegate(root, 'button', (node) => handled.push(node))`}
